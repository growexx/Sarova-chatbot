1️⃣ Which food article is driving the highest variance?
SELECT 
    article,
    STDDEV(qty) AS qty_variance
FROM purchase_data
WHERE category = 'FOOD'
GROUP BY article
ORDER BY qty_variance DESC
LIMIT 1;
2️⃣ Top 10 SKUs contributing highest spend
SELECT 
    article,
    SUM(net) AS total_spend
FROM purchase_data
GROUP BY article
ORDER BY total_spend DESC
LIMIT 10;
3️⃣ Top SKUs contributing 60% spend (Pareto)
WITH sku_spend AS (
    SELECT article, SUM(net) AS spend
    FROM purchase_data
    GROUP BY article
),
ranked AS (
    SELECT *,
           SUM(spend) OVER (ORDER BY spend DESC) /
           SUM(spend) OVER () AS cumulative_pct
    FROM sku_spend
)
SELECT *
FROM ranked
WHERE cumulative_pct <= 0.60;
4️⃣ Monthly growth rate of beverage purchases
WITH monthly AS (
    SELECT 
        DATE_TRUNC('month', date) AS month,
        SUM(net) AS monthly_spend
    FROM purchase_data
    WHERE category = 'BEVERAGE'
    GROUP BY month
)
SELECT 
    month,
    monthly_spend,
    (monthly_spend - LAG(monthly_spend) OVER (ORDER BY month)) /
    LAG(monthly_spend) OVER (ORDER BY month) AS growth_rate
FROM monthly
ORDER BY month;
5️⃣ Items with highest demand volatility
SELECT 
    article,
    STDDEV(qty) AS volatility
FROM purchase_data
GROUP BY article
ORDER BY volatility DESC
LIMIT 10;
6️⃣ Category with highest cost increase trend
WITH monthly_price AS (
    SELECT 
        category,
        DATE_TRUNC('month', date) AS month,
        AVG(ave_p_price) AS avg_price
    FROM purchase_data
    GROUP BY category, month
)
SELECT 
    category,
    (MAX(avg_price) - MIN(avg_price)) / MIN(avg_price) AS price_growth
FROM monthly_price
GROUP BY category
ORDER BY price_growth DESC;
7️⃣ Month with peak demand seasonality
SELECT 
    EXTRACT(MONTH FROM date) AS month,
    SUM(qty) AS total_qty
FROM purchase_data
GROUP BY month
ORDER BY total_qty DESC;
8️⃣ Articles with abnormal purchase spikes
(Z-score based)

WITH stats AS (
    SELECT 
        article,
        AVG(qty) AS avg_qty,
        STDDEV(qty) AS std_qty
    FROM purchase_data
    GROUP BY article
)
SELECT 
    p.article,
    p.date,
    p.qty,
    (p.qty - s.avg_qty) / s.std_qty AS z_score
FROM purchase_data p
JOIN stats s ON p.article = s.article
WHERE ABS((p.qty - s.avg_qty) / s.std_qty) > 3;
9️⃣ Procurement spend trend (24 months)
SELECT 
    DATE_TRUNC('month', date) AS month,
    SUM(net) AS total_spend
FROM purchase_data
GROUP BY month
ORDER BY month;
🔟 Year-over-Year beverage growth
WITH yearly AS (
    SELECT 
        EXTRACT(YEAR FROM date) AS year,
        SUM(net) AS yearly_spend
    FROM purchase_data
    WHERE category = 'BEVERAGE'
    GROUP BY year
)
SELECT 
    year,
    yearly_spend,
    (yearly_spend - LAG(yearly_spend) OVER (ORDER BY year)) /

1️⃣1️⃣ % of spend in top 20% SKUs
WITH sku_spend AS (
    SELECT article, SUM(net) AS spend
    FROM purchase_data
    GROUP BY article
),
ranked AS (
    SELECT *,
           NTILE(5) OVER (ORDER BY spend DESC) AS bucket
    FROM sku_spend

    SUM(spend) AS top_20pct_share
FROM ranked;
1️⃣2️⃣ Forecast vs actual variance (if forecast table exists)
SELECT 
    p.date,
    p.article,
    p.qty AS actual_qty,
    f.forecast_qty,
    (p.qty - f.forecast_qty) AS variance
FROM purchase_data p
JOIN forecast_data f
    ON p.date = f.date
    AND p.article = f.article

1️⃣3️⃣ Over / Under estimation detection
SELECT 
    article,
    AVG(p.qty - f.forecast_qty) AS avg_error
FROM purchase_data p
JOIN forecast_data f
    ON p.date = f.date
    AND p.article = f.article
GROUP BY article
ORDER BY avg_error DESC;
1️⃣4️⃣ Align procurement with occupancy growth
WITH monthly_purchase AS (
    SELECT DATE_TRUNC('month', date) AS month,
           SUM(net) AS total_spend
    FROM purchase_data
    GROUP BY month
),
monthly_occupancy AS (
    SELECT DATE_TRUNC('month', date) AS month,
           AVG(cf_occupancy) AS avg_occ
    FROM operational_data
    GROUP BY month
)
SELECT 
    p.month,
    p.total_spend,
    o.avg_occ
FROM monthly_purchase p
JOIN monthly_occupancy o
ON p.month = o.month
ORDER BY p.month;
1️⃣5️⃣ SKUs spiking during high-tourism months
WITH high_occ_months AS (
    SELECT DATE_TRUNC('month', date) AS month
    FROM operational_data
    GROUP BY month
    HAVING AVG(cf_occupancy) >
           (SELECT AVG(cf_occupancy) FROM operational_data)
)
SELECT 
    article,
    SUM(qty) AS total_qty
FROM purchase_data
WHERE DATE_TRUNC('month', date) IN (SELECT month FROM high_occ_months)
GROUP BY article
ORDER BY total_qty DESC;
1️⃣6️⃣ Items increasing in cost faster than consumption
WITH price_growth AS (
    SELECT 
        article,
        MAX(ave_p_price) - MIN(ave_p_price) AS price_change
    FROM purchase_data
    GROUP BY article
),
qty_growth AS (
    SELECT 
        article,
        MAX(qty) - MIN(qty) AS qty_change
    FROM purchase_data
    GROUP BY article
)
SELECT 
    p.article
FROM price_growth p
JOIN qty_growth q
ON p.article = q.article
WHERE p.price_change > q.qty_change;
